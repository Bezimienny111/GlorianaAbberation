#founding the republics
#batavian republic netherlands
event = {
	id = 120750
	trigger = {
		event = 3148
		owned = { province = 339 data = FRA }
		control = { province = 339 data = FRA }
		NOT = {
			exists = HOL
		}
	}
	random = no
	country = FRA
	name = "The Batavian Republic"
	desc = "Revolutionary tendencies spread after the fall of the French Bourbons and the founding of the First French Republic. French military gains by Napoleon Bonaparte and other commanders saw the founding of puppet Republics by France in her newly aquired territories. The Batavian Republic was one such puppet. It covered the old Kingdom of the Netherlands and would see Napoleons brother Louis Banaparte as its king when Napoleon became Emperor of France."
	style = 4

	date = { day = 1 month = january year = 1790 }
	offset = 10
	deathdate = { day = 1 month = january year = 1814 }

	action_a = {
		name = "Create the Republic"
		command = { type = independence which = HOL }
		command = { type = vp value = 10 }
		command = { type = sleepevent which = 3515 }
		command = { type = trigger which = 120753 }
		command = { type = trigger which = 120759 }
	}

	action_b = {
		name = "Rule it from Paris"
		command = { type = domestic which = centralization value = -1 }
		command = { type = vp value = 5 }
	}
}
