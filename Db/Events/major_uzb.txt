# The Khanate of Kokand #
event = {

	id = 50041
	trigger = {
		OR = {
			tag = BUK
			tag = KHI
			tag = UZB
		}
		atwar = no
		owned = { province = 1598 data = -1 } # Tadjikistan
		NOT = { exists = KOK }
	}
	random = no
	name = "The Khanate of Kokand"
	desc = "The Khanate of Kokand was established in 1709 when the Shaybanid emir Shahrukh, of the Ming Tribe of Uzbeks, declared independence from the Khanate of Bukhara, establishing a state in the eastern part of the Fergana Valley. He built a citadel as his capital in the small town of Kokand, thus starting the Khanate of Kokand. His son, Abdul Kahrim Bey, and grandson, Narbuta Bey, enlarged the citadel, but both were forced to submit as a protectorate, and pay tribute to, the Qing dynasty between 1774 and 1798. Though a small and fairly insular state, Kokand did later play an important role during the Afaqi Khoja rebellion in Dzungaria, before eventually being annexed into the Russian Empire as the Fergana Oblast in the 1870s."

	date = { day = 1 month = january year = 1710 }
	offset = 150
	deathdate = { day = 1 month = january year = 1837 }

	action_a ={	# Grant them their freedom #
		name = "Grant them their freedom"
		command = { type = cityname which = 1598 value = "Kokand" }
		command = { type = free which = KOK }
		command = { type = relation which = KOK value = 100 }
		command = { type = domestic which = CENTRALIZATION value = -1 }
		command = { type = domestic which = SERFDOM value = -1 }
		command = { type = stability value = 1 }
	}

	action_b ={	# Stamp out the Ming traitors! #
		name = "Stamp out the Ming traitors!"
		command = { type = revolt which = 1591 }
		command = { type = revolt which = 1598 }
		command = { type = revolt which = 1598 }
		command = { type = domestic which = CENTRALIZATION value = 1 }
		command = { type = domestic which = SERFDOM value = 1 }
		command = { type = stability value = -2 }
	}
}
