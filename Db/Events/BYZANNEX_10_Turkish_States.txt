#(1419-1500) Submission of the Sultan - Byzantium
event = {
	id = 3001173
	trigger = {
		vassal = { country = BYZ country = TUR }
		NOT = { war = { country = TUR country = BYZ } }
	}
	random = NO
	country = BYZ
	name = "Submission of the Sultan"
	desc = "The Roman Empire, after a miraculous campaign of reconquest against the Turk, succeeds at last in seizing the enemy capital at Bursa. The Ottoman Sultan submits himself to the mercy of the Emperor and offers his realm to the New Rome."
	date = { day = 1 month = january year = 1337 }
	offset = 30 #1 month
	deathdate = { year = 1500 }

	action_a = {
		name = "Accept the Turks' submission"
		command = { type = treasury value = 150 }
		command = { type = trigger which = 3001174 } #TUR: Submission of The Sultan
		#command = { type = vp value = 10 }
	}
	action_b = {
		name = "Let the walking dead alone"
		command = { type = badboy value = -2 }
		command = { type = treasury value = 250 }
		command = { type = INf which = -2 value = 5000 }
		command = { type = INf which = -2 value = 3000 }
	}
}
