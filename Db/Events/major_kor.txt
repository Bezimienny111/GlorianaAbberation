	id = 3890
	random = no
	country = KOR
	name = "EVENTNAME3890"
	desc = "EVENTHIST3890"
	style = "3890"

	date = { day = 1 month = june year = 1446}
	offset = 180

	action_a ={#Replace the Chinese Characters with Hangul
		name = "ACTIONNAME3890A"
		command = { type = domestic which = serfdom value = -2 }
		command = { type = domestic which = innovative value = 2 }
		command = { type = domestic which = aristocracy value = -1 }
		command = { type = domestic which = centralization value = -1 }
		command = { type = treasury value = -200}
		command = { type = provincetax which = 642 value = 2}
		command = { type = stability value = -1}
		command = {type = relation which = CHI value = -50}
	}
	action_b ={#Keep Chinese Characters
		name = "ACTIONNAME3890B"
		command = { type = stability value = -1}
	}
}
