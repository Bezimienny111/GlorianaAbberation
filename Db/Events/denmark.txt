	id = 104001
	random = no
	country = DAN
	name = "The Crowning of Valdemar Atterdag"
	desc = "With the assassination of Gerhard III by Danish patriot Niels Ebbesen on April 1, 1340, the eight year interregnum in Denmark was almost at an end. Valdemar Atterdag, the son of the last Danish monarch Christopher II, had waited as a claimant to the throne since his father's death in 1332. The power vacuum created with the removal of the German tyrant Gerhard gave Valdemar the much awaited opportunity to make good on this claim. Valdemar made an arrangement with Count Johann the Mild of Holstein, in which Valdemar would be crowned King of Denmark, while Holstein was to gain economically. This arrangement was, however, the effective end of German power in Denmark. Valdemar Atterdag, now Valdemar IV used his foothold and this new received legitimacy to consolidate his power throughout all of the Danish lands, beginning a path that would reestablish the strength of Denmark, and eventually lead to the Kalmar Union."

		command = { type = breakvassal which = SHL }
		command = { type = domestic which = centralization value = 1 }
		command = { type = treasury value = -150 }
	}
}
