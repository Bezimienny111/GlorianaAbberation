#(1470-1550) The Persian Rebellion
event = {
	id = 3001445
	trigger = {
		event = 3001112 #BYZ: Office of Barbarians
		exists = PER
	}
	random = NO
	country = BYZ
	name = "The Persian Rebellion"
	desc = "Our spies and merchants bring word that a vast rebellion has broken out in the lands to our East. Apparently, the Muhammadan heresy has heresies of its own - and one is rising strongly against its lords. Should we support them?"

	date = { year = 1470 }
	offset = 3600 #10 years
	deathdate = { year = 1550 }

	action_a = {
		name = "Is the Pope Catholic?"
		command = { type = treasury value = -400 }
		command = { type = relation which = PER value = 300 }
		command = { type = relation which = AKK value = -100 }
		command = { type = relation which = QAR value = -100 }
		command = { type = trigger which = 3001446 }
	}
	action_b = {
		name = "A little, perhaps."
		command = { type = treasury value = -50 }
		command = { type = relation which = PER value = 20 }
		command = { type = sleepevent which = 3001447 }
	}
	action_c = {
		name = "Rebels? Heretics? Have you no honour, good sir?"
		command = { type = stability value = 1 }
		command = { type = relation which = AKK value = 50 }
		command = { type = relation which = QAR value = 50 }
		command = { type = relation which = IRA value = 50 }
		command = { type = relation which = MOG value = 50 }
		command = { type = sleepevent which = 3001447 }
	}
}
