event = {

	id = 131100
	trigger = {
          	 AND = {
                 	OR = {
                       		owned = { province = 404 data = -1 }

                  	}
                  	OR = {
                        	owned = { province = 402 data = -1 }

                        	owned = { province = 400 data = -1 }
                        	vassal = { country = MLO country = PIS }

                       		owned = { province = 390 data = -1 }
                        	vassal = { country = MLO country = MAN }

			atwar = no

				exists = ITA
			}
           	}

	random = no
	country = MLO
	name = "Consolidation of Northern Italy"
	desc = "Trying to unite Italy had been on the minds of many of the rulers of the Italian states as well as major European powers who did everything in their power to prevent it. It was finally achieved in 1861 when the great powers were distracted by the Franco-Prussian war. There were many things that were needed to be done before an united Italy could burst itself onto the international scene, gaining the necessary territory was only but one of those requisites needed. However we have a great powerbase in Northern Italy, we could start the process to unite Italy my lord."
	style = 2

	date = { day = 1 month = january year = 1337 }
	offset = 90
	deathdate = { day = 31 month = december year = 1820 }

	action_a = {
		name = "Start the process to Unite Italy"
		command = { type = vp value = 5 }
		command = { type = stability value = -1 }
		command = { type = trigger which = 131101 } #savoy chooses to accept
		command = { type = trigger which = 131102 } #modena chooses to accept
		command = { type = trigger which = 131103 } #florence chooses to accept
		command = { type = trigger which = 131104 } #siena chooses to accept
		command = { type = trigger which = 131105 } #mantua chooses to accept
		command = { type = trigger which = 131116 } #papal resistance or support
		command = { type = setflag which = mloitaly }
	}

	action_b = {
		name = "Let the idea of a United Italy die forever"
		command = { type = relation which = HAB value = 100 }
		command = { type = relation which = FRA value = 50 }
		command = { type = relation which = SPA value = 50 }
		command = { type = domestic which = aristocracy value = -1 }
	}
}
