# 3009400: Wielka Noc Odwetu na Groblach Tenochtitlan
event = {
	id = 3009400
	random = no
	country = AZT
	name = "Wielka Noc Odwetu na Groblach Tenochtitlan"
	desc = "Gdy hiszpanscy konkwistadorzy probowali wymknac sie pod oslona nocy z Tenochtitlan obciazeni zagrabionym zlotem, wojownicy Mexicow przerwali mosty i zaatakowali z setek wojennych czolen. Zamiast hekatomby, bitwa konczy sie calkowitym zniszczeniem najezdzcow i pojmaniem zywcem hiszpanskich rusznikarzy, kowali oraz ich koni bojowych!"

	trigger = {
		owned = { province = 27 data = -1 } # Tenochtitlan
		control = { province = 27 data = -1 }
		year = 1515
		not = { year = 1540 }
		not = { hasflag = { which = azt_captured_technology value = 1 } }
	}

	date = { day = 1 month = january year = 1515 }
	offset = 60
	deathdate = { day = 30 month = december year = 1540 }

	action_a = {
		name = "Uderzcie ze wszystkich stron i pojmijcie kowali zywcem!"
		command = { type = treasury value = 300 }
		command = { type = land_morale value = 0.20 }
		command = { type = land value = 500 }
		command = { type = setflag which = azt_captured_technology value = 1 }
		command = { type = stability value = 1 }
		command = { type = vp value = 100 }
	}
}
