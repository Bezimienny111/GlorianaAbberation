# The End of the Order #
event = {
	id = 50056
	trigger = {
		event = 3540
		event = 3145
		war = { country = FRA country = KNI }
	}
	random = no
	country = KNI
	name = "The End of the Order"
	desc = "The chaos of the French Revolutionary Wars brought calamity to the Order of St. John, which for decades had struggled to finance its operations and repair its crumbling citadel on Malta. By the turn of the 19th century the tithes from its various branches across Europe had all but dried up, and the Order was so poor and dilapidated that it even struggled to find candidates for election as Grand Master - a position once considered highly coveted. When Napoleon decided to invade Malta en route to Egypt during his 1798 campaign against the Ottomans, Valletta's antiquated medieval defences did little to hinder the well-disciplined troops of Revolutionary France and their modern heavy artillery. Sensing the gravity of the situation, in September 1806 the Swedish government offered the sovereignty of the island of Gotland to the Order."
	
	date = { day = 1 month = january year = 1792 }
	offset = 30
	deathdate = { day = 1 month = january year = 1815 }
	
	action_a ={ # Defend Malta to the last man! #
		name = "Defend Malta to the last man!"
		command = { type = fortress which = 819 value = 1 }
		command = { type = gainbuilding which = 819 value = barrack }
		command = { type = domestic which = OFFENSIVE value = -2 }
		command = { type = relation which = FRA value = -100 }
		command = { type = land value = 500 }
		command = { type = naval value = 500 }
		command = { type = manpower value = 25 }
		command = { type = treasury value = -500 }
		command = { type = stability value = 1 }
	}

	action_b ={ # Flee to Gotland #
		name = "Flee to Gotland"
		trigger = { 
			exists = SWE
			owned = { province = 308 data = SWE }
			NOT = {
				war = { country = SWE country = KNI }
				alliance = { country = SWE country = FRA }
			}
		}
		command = { type = breakvassal which = -1 }
		command = { type = trigger which = 50057 }
		command = { type = domestic which = INNOVATIVE value = 1 }
		command = { type = diplomats value = 3 }
		command = { type = stability value = -1 }
	}
}
