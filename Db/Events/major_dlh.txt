# Tiger on the Border
event = {
	id = 3824
	trigger = {
		exists = MOG

		owned = {
			province = 1523 # Thar
			data = -1
		}

		NOT = {
			war = { country = DLH country = MOG }
		}
	}
	random = no
	country = DLH
	name = "EVENTNAME3824"
	desc = "EVENTHIST3824"

	date = { day = 1 month = january year = 1510 }
	offset = 30
	deathdate = { day = 1 month = january year = 1530 }

	action_a = { # Negotiate with Babur
		name = "ACTIONNAME3824A"
		command = { type = cedeprovince which = MOG value = 1522 } #Bikaner
		command = { type = cedeprovince which = MOG value = 1523 } #Thar
		command = { type = relation which = MOG value = 50 }
		command = { type = casusbelli which = MOG value = 240 }
		command = { type = stability value = 1 }

		command = { type = relation which = MOG value = -200 }
		command = { type = war which = MOG }
		command = { type = stability value = -4 }
	}
}
