#Hansa factional dominance#
event = {

	id = 100001
	random = no
	country = HSA
	name = "Hansa factional dominance"
	desc = "The Hansa was not a country per se, but was an amalgamation of trading cities. Each of these cities had a representative which represented the cities interests in the Hansatag (council). Every so often different factions would hold a dominance within the council, but which one?"
	style = 1

	date = { day = 1 month = january year = 1419 }
	offset = 90
	deathdate = { day = 31 month = december year = 1428 }

		ai_chance = 25
		name = "The Luft faction will have dominance"
