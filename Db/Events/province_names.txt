event = {

	id = 160000
	trigger = {
		provinceculture = { province = 65 data = anglosaxon }
		owned = { province = 65 data = ENG }
	}

	random = no
	country = ENG
	name = "Stamp Our Mark - Catawba"
	desc = "We have placated the locals in this area and the land is ours. We should give the main town in this region a suitable name."

	date = { day = 1 month = january year = 1419 }
	offset = 10
	deathdate = { day = 1 month = january year = 1820 }

	action_a = {
		name = "Make it so"
		command = { type = cityname which = 65 value = "Lynchburg" }
		command = { type = treasury value = -1 }
	}

	action_b = {
		name = "No, I like the name"
		command = { }
	}
}
