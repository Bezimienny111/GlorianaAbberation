#Qara Koyunlu continuing the attack?
event = {

	id = 150100
	trigger = {
		NOT = {
			relation = { country = ABB data = 0 }
			alliance = { country = MAM country = ABB }
		}
	}
	random = no
	country = QAR

	name = "Jalayirid Treachery"
	desc = "Ahmad of the Jalayrids broke his bargain with Qara Koyunlu by invading Azerbaijan, briefly occupying Tabriz until Qara Koyunlu defeated the Jalayirids in August 1410. Ahmad was either captured and executed or assassinated. Yusuf named his son Shah Muhammad to rule in Baghdad in 1412 AD. The remnants of the Jalayirid dynasty were pushed south into lower Iraq, ruling over the towns of Al-Hillah, Wasit, and Basra until extinguished by the Qara Qoyunlu in 1432. Will we complete the extinction by wiping out Ahmad's heirs?"
	style = 0

	date = { day = 1 month = january year = 1419 }
	offset = 10
	deathdate = { day = 1 month = january year = 1517 }

	action_a = {
		name = "Yes, Ahmad's sins must be punished to its logical conclusion"
		command = { type = casusbelli which = ABB value = 240 }
		command = { type = relation which = ABB value = -200 }
	}

		name = "No, they are not worth it. We have bigger fish to fry"
		command = { type = relation which = ABB value = 50 }
		command = { type = stability value = 1 }
	}
}
