#(1420-1430) Hesychasm flavor event (by Driftwood)
event = {
	id = 3001488
	trigger = {
		religion = orthodox
	}
	random = NO
	country = BYZ
	name = "Hesychasm returns!"
	desc = "The Byzantines were always fond of religious fights, spending centuries to hammer out the exact interrelationships between the components of the Trinity, before moving on to the desirability of icons, back to the components of the Trinity, and so on. Of course, these debates usually became foils for political struggles of a secular nature after a couple of years. One such struggle was sparked by the practice of the monks of Mt. Athos, who would hold their breath and meditate, sometimes being blessed by a vision of the light of God. The Patriarch interpreted this as a heretical belief in an uncreated Light, a fourth component of the Godhead. More generally, it reflected a Byzantine tendency towards mysticism over intellectualism. Though settled in the 14th century, the debate was tied up with that over the Union with the Catholic Church and periodically burst back into contention again."

	date = { year = 1337 }
	offset = 3600 #10 years
	deathdate = { year = 1430 }

	action_a = {
		name = "Endorse the Hesychasts"
		command = { type = domestic which = INNOVATIVE value = -1 }
		command = { type = stability value = 1 }
		command = { type = relation which = PAP value = -50 }
		command = { type = revolt which = -1 }
	}
	action_b = {
		name = "Tolerate the Hesychasts"
		command = { type = domestic which = SERFDOM value = -1 }
		command = { type = stability value = -2 }
	}
	action_c = {
		name = "Persecute the Hesychasts"
		command = { type = domestic which = CENTRALIZATION value = 1 }
		command = { type = domestic which = INNOVATIVE value = 1 }

	name = "Georgios Skholarios"
	desc = "Historically a Byzantine scholar and the Patriarch Gennadios B', Georgios trained as a lawyer and became a Judge-General in charge of the University. Learning Latin, he read and admired the works of Hagios Thomas Aquinas and wrote a number of philosophical works. A delegate to the Council of Florence and pro-unionist, he had second thoughts after his return to Constantinople. This was partially because he believed the end of the world was at hand: by Byzantine calculations the world would be 7000 years old in 1492, a turning point and certainly the Anti-Christ was at the gates. Therefore, to his mind, it was more important to keep the Faith pure than preserve the worldly Empire. In this capacity he was instrumental in establishing a constitution with the Sultan preserving the structure of the Church. With the Turks kept at bay and routed, his patriarchate is more secure and long-lived and his consummate skill helps to spread order and peace to the new-won lands."

		command = { type = infra value = 50 }
		command = { type = stability value = 2 }
	}
}
