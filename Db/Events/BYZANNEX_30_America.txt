#(1770-1790) American Declaration of Independence
event = {
	id = 3001418
	trigger = {
		exists = USA
		domestic = { type = serfdom value = 5 }
	}
	random = NO
	country = BYZ
	name = "American Declaration of Independence"
	desc = "Representatives of England's North American colonies have gathered together and signed a document seceding their provinces from the Crown. An emissary, John Jay, has been sent to the Imperial court to try to garner recognition, funding and supplies for use against England. Should we interfere?"

	date = { year = 1770 }
	offset = 30
	deathdate = { year = 1790 }

	action_a = {
		name = "Support the rebellion"
		command = { type = treasury value = -600 }
		command = { type = stability value = -2 }
		command = { type = domestic which = ARISTOCRACY value = -1 }
		command = { type = relation which = USA value = 200 }
		command = { type = relation which = ENG value = -200 }
		command = { type = casusbelli which = ENG value = 60 }
		command = { type = trigger which = 3001420 }
	}
	action_b = {
		name = "Support England against the rebels"
		command = { type = stability value = 1 }
		command = { type = domestic which = ARISTOCRACY value = 1 }
		command = { type = relation which = ENG value = 100 }
		command = { type = casusbelli which = USA value = 60 }
		command = { type = relation which = USA value = -400 }
	}
	action_c = {
		name = "Do business with both sides"
		command = { type = treasury value = 100 }
		command = { type = relation which = USA value = -50 }
		command = { type = relation which = ENG value = -50 }
	}
}
