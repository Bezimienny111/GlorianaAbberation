			war = { country = MCH country = CHG }
		}
	}
	random = no
	country = MCH
	name = "The Conquest of Dzungaria"
	desc = "The conquest of Dzungaria took place in the 18th century when the Qing Dynasty of China sought to extend its authority and control over the Dzungar Khanate, a Mongol state located in the former Chagatai lands. The conflict began in the early 1750s and culminated in the Qianlong Emperor's campaign against the Dzungars from 1755 to 1759.
