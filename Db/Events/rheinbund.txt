#french proclaim the rheinbund
event = {
	id = 120800
	trigger = {
		OR = {
			AND = {
				event = 3148
				owned = { province = 345 data = FRA }
				control = { province = 345 data = FRA }
				NOT = {
					exists = RHB
					}
				}
			AND = {
				event = 3148
				owned = { province = 346 data = FRA }
				control = { province = 346 data = FRA }
				NOT = {
					exists = RHB
					}
				}
			}
		}
	random = no
	country = FRA
	name = "The Confederation of the Rhine"
	desc = "Napoleon created the Confederation of the Rhine after his military sucesses in Germany in 1806. After the defeat of Prussia in the same year more member states threw their lot in with the Confederation. The Prince President of the Federation was the former Archbishop of Mainz Karl Theodor Anton Marie von Dalberg. The Confederation would last until the defeat of Napoleon around 1813-4 when many of the states turned and joined the allies. Nevertheless, the Treaty that ended the war in 1815 after the hundred days would see the German states borders pretty much fixed at those of the member states under the Confederation."
	style = 4

	date = { day = 1 month = january year = 1806 }
	offset = 15
	deathdate = { day = 1 month = january year = 1821 }

	action_a = {
		name = "Create the Confederation"
		command = { type = independence which = RHB }
		command = { type = vp value = 10 }
		command = { type = trigger which = 120842 }
	}

	action_b = {
		name = "Rule it from Paris"
		command = { type = domestic which = centralization value = -1 }
		command = { type = vp value = 5 }
	}
}
