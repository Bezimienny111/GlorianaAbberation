	id = 105587
	trigger = {
		NOT = {

		} 
	}
	random = no
	country = ENG
	name = "John Wyclif's theories"
	desc = "In 1373, John Wyclif, a doctor of theology from Oxford, began writing ‘De Domino Divinio’ in which he contested Church traditional authority. Wyclif criticized Church corruption under the Avignon pope and event went so far as to propose a random election for the Pope. Two years later he composed ‘De Civili Dominio’ in which he rejected the notion that the Church has any sort of temporal power. While the Church was horrified by his tracts, and demanded censure, he found a receptive audience not only among those at the universities, but also with the nobility who agreed with his criticism of the worldly possessions of the clergy. He was summoned in 1377 and 1378 by the Bishop of London, Guillaume de Courtenay, to explain his views, and although a papal bull was written against him, his relations with John of Gaunt saved him from any serious repercussions. His reputation was so great at that moment that both Parliament and Londoners protected him against the Curia."

	date = { day = 1 month = october year = 1377 }
	offset = 500
	deathdate = { day = 1 month = october year = 1380 }

	action_a = {
		name = "He raises valid points"

		command = { type = relation which = PAP value = -50 }
	}
	action_b = {
		name = "Condemn him as heretic immediately!"
		command = { type = stability value = -2 }
		command = { type = domestic which = innovative value = -1 }
		command = { type = revoltrisk which = 36 value = 2 }
		command = { type = revolt which = 247 }
