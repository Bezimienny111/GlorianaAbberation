#(1424) Venice Gains Thessaloniki
event = {
	id = 3001091
	trigger = {
		owned = { province = 358 data = TUR } #Macedonia
	}
	random = NO
	country = VEN
	name = "Venice Gains Thessaloniki"
	desc = "Murad rapidly put down Mustafa's revolt and angrily attacked Byzantium. Manuel released another pretender, also named Mustafa, also Murad's brother, but this one fared no better. In 1423, the Turks invaded Morea, smashing through the Hexamilium (6 mile wall) that Manuel II had built across the Isthmus of Corinth for protection in 1414. They also pressed Thessaloniki, still a Byzantine city, so hard that the governor turned it over to the Venetians and wished them better luck. By 1424, with no help forthcoming, the Byzantines sadly accepted vassalage to the Turks. Venice held onto Thessaloniki until 1430."

	date = { year = 1424 }
	offset = 30 #1 month
	deathdate = { year = 1425 }

	action_a = {
		name = "I'll take it while I can" # Take control of Thessalonica
		command = { type = addcore_national which = 358 } #Macedonia
		command = { type = provincetax which = 370 value = 1 } #Veneto
		command = { type = treasury value = 100 }
		#command = { type = vp value = 10 }
	}
	action_b = {
		name = "Hand it over to the Turks" # Hand it over to the Turks
		command = { type = relation which = TUR value = 100 }
		command = { type = sleepevent which = 3001092 } #VEN: Fall of Thessaloniki
		command = { type = sleepevent which = 3001093 } #ALL: Recovery of Thessaloniki
	}
}
