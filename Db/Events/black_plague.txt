# Kaffa

event = {
id = 2200001
random = no
province = 464
name = "Black plague strikes Kaffa!"
desc = "The Black Plague originated with the fleas in little rodents such as 
rats, squirrels, and mice on the northern coast of the Black Sea, where the 
Italians had trading colonies by 1346. In fact, the Black Plague spread 
through the importing and exporting of goods from country to country and it 
is known that it started in Kaffa in the autumn of 1346. Nobody ever thought 
that the fleas on the rodents were the cause of the deadly virus. Often, in 
the winter time, the plague would come about because the little rodents 
would look for a warm place and would live in people's houses. The whole 
family would then be infected with the plague, and sooner or later the whole 
village or town would be infected . The Black Plague was responsible for 
millions of deaths. Most of the deaths recorded were in Europe."


style = 3
date = { day = 25 month = september year = 1346 }
offset = 5

}
}
