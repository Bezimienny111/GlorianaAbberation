			exists = PRU
			}
		}
	random = yes
	country = HAB
	name = "German Resistance"
	desc = "The Holy Roman Empire relied on the principles of independent electors who would vote for, and elect, an Emperor. In this manner the Empire was not a dynastic institution. 	style = 1

	action_a = {
		name = "release the hounds!"
		command = { type = free which = PRU }
		command = { type = revolt which = -1 }
	}
}
