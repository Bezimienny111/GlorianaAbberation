# Event 3009010: POR - The Flight of the Court to Brazil
# Trigger: war with BYZ, BYZ controls Lisboa (441)
#-----------------------------------------------------------------------------

	id = 3009010
	random = no
	country = POR

	name = "The Flight of the Portuguese Court"
	desc = "Lisbon has fallen! The triumphant legions of the Eastern Roman Emperor occupy the Tagus estuary and the ancient royal citadel. King and court face imminent capture unless they take to the Atlantic. Our transatlantic fleet stands ready to carry the Braganza dynasty and the royal archives across the ocean to our flourishing Brazilian captaincies or our island strongholds."
	style = 1

	date = { day = 1 month = january year = 1337 }
	offset = 15
	deathdate = { day = 30 month = december year = 1838 }

	trigger = {
		war = { country = POR country = BYZ }
		control = { province = 441 data = BYZ } # Tago / Lisboa controlled by BYZ
		NOT = { flag = por_capital_relocated }
		NOT = { capital = 209 }
