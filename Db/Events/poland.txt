#(1338) The Second Conference of Visegrad
#we're in a bad need of an alternative dynasty in case Angevins don't ascend the throne.
event = {
	id = 115000
	trigger = {
		exists = HUN
		NOT = { war = { country = POL country = HUN } }		
	}
	random = no
	country = POL
	name = "The Second Conference of Visegrad"
	desc = "In 1338, a conference of monarchs took place in the Hungarian city of Visegrad. Taking part were the kings: Kazimierz of Poland , Karl Robert of Hungary, and Jan of Bohemia. At the first meeting, Jan had given up his claims on the Polish throne in exchange for Kazimierz giving up his claims on the Silesian duchies. The second meeting's main point was to determine the Angevin succession to the throne of Poland, in the event of Kazimierz dying without issue."

	date = { day = 1 month = february year = 1338 }
	offset = 30
	deathdate = { day = 30 month = december year = 1338 }

	action_a ={
		trigger = {
			NOT = { war = { country = HUN country = POL } }
		}
		name = "The Angevins will rule Poland"
		command = { type = stability value = 1 }
		command = { type = dynastic which = HUN }
		command = { type = setflag which = [hungarian_succession] }
		command = { type = relation which = HUN value = 50 }
		command = { type = relation which = BOH value = 30 }
		command = { type = badboy value = -1 } 
		command = { type = trigger which = 117001 } #BOH: The Second Conference of Visegrad
		command = { type = trigger which = 118002 } #HUN: The Second Conference of Visegrad
	}
	action_b ={
		name = "Why should an Angevin take our throne?"
		command = { type = stability value = -2 }
		command = { type = breakdynastic which = HUN }
		command = { type = relation which = HUN value = -50 }
		command = { type = relation which = BOH value = -30 }
		command = { type = badboy value = 2 } 
		command = { type = trigger which = 118003 } #HUN: The Second Conference of Visegrad
		command = { type = setflag which = [no_succession] }
	}
}

	id = 115002

		NOT = { 
			war = { country = POL country = LAT }

			alliance = { country = POL country = LAT } 
		}
		exists = LAT
	}
	random = no
	country = POL
	name = "Peace of Kalisz"
	desc = "The conflict between Poland and the Teutonic Order dates back to 1308, when the Teutons, after asked by the Poles for assistance in defending Danzig against the Brandenburgians, took the city and massacred part of its population. They defended their actions by stating that Polish prince, Wladyslaw Lokietek, had not paid them for their help. All of Pomerelia soon fell into their hands, which resulted in a long state of conflict and many short wars between Poland and the Order, one of which (1329-1332) led almost to the collapse of Wladyslaw's reign. His son, Kazimierz III sought to improve relations with his neighbours and arranged a 'perpetual peace' with the Teutons. Under the terms of the treaty, Danzig would be recognized as the Order's 'perpetual alms' from the Polish king, for which the Order would hand over Cujavia and Dobrinia. The treaty was signed in Kalisz on 8 July, 1343."

	date = { day = 8 month = june year = 1343 }
	offset = 30
	deathdate = { day = 30 month = december year = 1350 }

		name = "Peace will strengthen us"
		command = { type = relation which = LAT value = 150 }
		command = { type = dynastic which = LAT }
		command = { type = cedeprovince which = LAT value = 301 }
		command = { type = provincetax which = 300 value = 2 } #Cujavia and Dobrinia lie in northern part of Wielkopolska prov on vanilla map; this is to symbolise their inclusion into Poland proper
	}
}
