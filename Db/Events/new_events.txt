# Pirates sack our capital! #
event = {
	id = 25009
	trigger = {
		NOT = { countrysize = 3 }
	}
	random = no
	persistent = yes

	name = "Pirates sack our capital!"
	desc = "Sire, pirate raiders have taken our capital and demand an outrageous ransom in exchange for their withdrawal. Shall we pay or not?"

	action_a = {	# Pay them off #
		name = "Pay them off"
		command = { type = treasury value = -100 }
		command = { type = disbandrebels which = -2 }
		command = { type = control which = -2 value = -1 }
		command = { type = infantry which = -2 value = 5000 }
	}

	action_b = {	# We can handle this #
		name = "We can handle this"
		command = { type = vp value = 10 }
	}
}

# Mongols! #
event = {
	id = 25020
	trigger = {
		OR = {
			tag = CHG # Chagatai
			tag = CHI # China
			tag = MCH # Manchu
			tag = MNG # Ming
			tag = QNG # Qing
			tag = SNG # Song
			tag = DHN # Han
			tag = FUJ # Fujiang
			tag = GUA # Guangdong
			tag = GXI # Guangxi
			tag = HEN # Henan
			tag = HNA # Hanan
			tag = HSI # Xsia
			tag = LIA # Liang
			tag = SXA # Shaanxi
			tag = SXI # Shanxi
			tag = TIA # Tianwen
			tag = TWN # Tiawan
			tag = XIN # Xianning
			tag = YUN # Yunnan
			tag = ZHE # Zhennang
			tag = ZHO # Zhou
			neighbour = NYN
	}
		exists = NYN
		NOT = { relation = { country = NYN data = 100 } }
		NOT = { year = 1600 }
	}
	random = yes

	name = "EVENTNAME25020"
	desc = "EVENTHIST25020"

	action_a = {	# Damn! #
		name = "ACTIONNAME25020A"
		command = { type = revolt which = -1 }
		command = { type = revolt which = -4 }
		command = { type = desertion which = -1 value = 5000 }
		command = { type = desertion which = -4 value = 2500 }
		command = { type = relation which = NYN value = -50 }
		command = { type = stability value = -2 }
	}

	action_b = {	# Pay them off #
		name = "ACTIONNAME25020B"
		command = { type = cash value = -250 }
		command = { type = relation which = NYN value = 50 }
		command = { type = stability value = -1 }
	}
}
