#ulster and the reformation#
event = {

	id = 120680
	trigger = {
		event = 100
		exists = HOL
		}
	random = no
	country = ULS
	name = "Ulaidh supports Protestant Cause"
	desc = "With the reformation, Ulsterian nobles threw their lot in with the Protestant cause, which would see them turn on their Catholic brothers. Protestant Ulaidh became a bulwark for the Protestant English and would cement the English Queen Elizabeth I as Queen of Ireland."
	style = 2

	date = { day = 1 month = january year = 1550 }
	offset = 30
	deathdate = { day = 1 month = january year = 1650 }

	action_a = {
		name = "Protestant Rights!"
		command = { type = religion which = protestant }
		command = { type = conversion which = -2 }
		command = { type = conversion which = -1 }
		command = { type = conversion which = -1 }
		command = { type = religiousrevolt which = -1 }
		command = { type = dynastic which = HOL }
		command = { type = dynastic which = ENG }
		command = { type = relation which = ENG value = 100 }
		command = { type = relation which = HOL value = 50 }
		command = { type = relation which = EIR value = -100 }
		command = { type = relation which = LEI value = -100 }
		command = { type = relation which = MEA value = -100 }
		command = { type = relation which = MST value = -100 }
		command = { type = relation which = CON value = -100 }
		command = { type = stability value = 2 }
	}

	action_b = {
		name = "It is not in our interests"
		command = { type = relation which = EIR value = 50 }
		command = { type = relation which = LEI value = 50 }
		command = { type = relation which = MEA value = 50 }
		command = { type = relation which = MST value = 50 }
		command = { type = relation which = CON value = 50 }
	}
}
