# Black Plague outbreak in Europe

event = {
	id = 200766
	trigger = {
		continent = Europe
		year = 1350
		NOT = { year = 1399 }
	}
	random = yes
	name = "Black Plague outbreak"
	desc = "The Black Plague returned periodically, striking mostly children, until it disappeared from Europe in 1399, not to return again until the 17th century. The history of the plague is a graphic example of how a disease can spread rapidly across amazing distances at an alarming rate. Besides, the plague spared no one. Doctors and clergy were usually among those who were hardest hit as they attempted to care for the dying."

	action_a ={
		name = "Darn..."
		command = { type = population which = -1 value = -700 }
		command = { type = cash value = -20 }
		command = { type = desertion which = -3 value = 350 }
		command = { type = stability value = -1 }
	}
}

### Nile floods ###

# Cataract

event = {
	id = 200751
	random = yes
	province = 743
	name = "Nile floods Cataract"
	desc = "For thousands of years the Nile River overflowed its banks every spring as the snow in the mountains transformed into a torrent of water. The annual floods carried dead and decaying plants in its muddy solution. The muddy water is called silt, and silt creates excellent farmland. Like all other natural things the flood was unpredictable at times and the amount of flooding was very important. Too much flooding and the crops would be planted late which would adversely affect them. Too little flooding and not enough area would be fertilized with the silt, which would mean poor crops in those areas if they were even planted at all."

	action_a ={
		name = "Darn"
		command = { type = population which = 743 value = -250 }
		command = { type = infra value = -50 }
		command = { type = trade value = -50 }
		command = { type = cash value = -25 }
	}
}
