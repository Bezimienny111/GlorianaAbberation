#random trading straits - gibralter#
event = {

	id = 120700
	trigger = {
		owned = { province = 445 data = -1 }
		owned = { province = 732 data = -1 }
		}
	random = yes
	name = "We control the Gibralter Strait"
	desc = "With control of the Gibralter Strait a nation could greatly influence the
traffic of trade and merchants that passed from the Mediterranean Sea into the Atlantic 
Ocean. Nations could profit greatly from the strategic importance of these straits, what is thy
will sire?"
	style = 0

	action_a = {
		name = "Tax all trade coming through"
		command = { type = merchants value = 3 }
		command = { type = naval value = 50 }
		command = { type = relation which = -1 value = -20 }
		command = { type = provincetax which = -2 value = 2 }
		command = { type = domestic which = mercantilism value = 1 }
	}

	action_b = {
		name = "Limited taxes on trade"
		command = { type = merchants value = 1 }
		command = { type = relation which = -1 value = 10 }
		command = { type = relation which = -1 value = 10 }
		command = { type = relation which = -1 value = -10 }
		command = { type = provincetax which = -2 value = 1 }
		command = { type = trade value = 25 }
	}

	action_c = {
		name = "Allow Free Trade through strait"
		command = { type = merchants value = -4 }
		command = { type = relation which = -1 value = 20 }
		command = { type = relation which = -1 value = 20 }
		command = { type = relation which = -1 value = 20 }
		command = { type = trade value = 50 }
		command = { type = domestic which = mercantilism value = -1 }
		command = { type = provincetax which = 445 value = 1 }
	}
}
