#Revolters 1.3

#Information on how countries are created from revolt or as vassals.
#As of 1.3 vassals can not be released whilst at war.

#date = { }	  First date of possible action.
#expirydate = { }    Last date of possible action.
#revolt = yes/no     Can revolt or not.
#annexable = yes/no  If yes, the country can always be annexed in one war even if not an annexable religion.
#no = <tag> or { }   Cannot revolt from the listed countries. Entry can be used several times. In this case, tags will be added to the list.
#yes = <tag> or { }  Can revolt from the listed countries only. Entry can be used several times. In this case, tags will be added to the list.
#vassal = yes/no     Can be released as vassal or not.
#minimum = { }       List of any mandatory revolted provinces for the country to appear as a revolter / list of all provinces given to the vassal if owned. Listed provinces are CORES of the new country.
#extra = { }         List of provinces that will join a revolter if also revolted. Listed provinces are CLAIMS of the new country.
#capital = <id>      Capital province. If the province is not specified, capital will be located in the "best" province according to accepted culture, religion, fortress level and city level, in descending order and as a combination. Capital can thus be located outside minimal provinces.
#group = <name>      Techgroup (same as the involved country if not specified).
#culture = <name>    Primary culture (same as the capital location if not specified).
#religion = <name>   State religion (same as the capital location if not specified).
#ai = "<name>"       AI file for the country (file is "default_ai.txt" if not specified).
